cmake_minimum_required(VERSION 3.22.1)
project(dictionary_storage_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(../../../main/cpp/hoshidicts hoshidicts)
add_executable(storage_tests storage_tests.cpp)
target_link_libraries(storage_tests PRIVATE hoshidicts)

    target_include_directories(legacy_import PRIVATE ../../../main/cpp/hoshidicts/src)
    target_link_libraries(legacy_import PRIVATE hoshidicts glaze::glaze libzstd_static unordered_dense)
